        // 标语切换系统
        // ==========================================
        function initSloganSystem() {
            sloganItems = document.querySelectorAll('.slogan-item');
            sloganItems[0].classList.add('active');
            lastSloganSwitchTime = Date.now();
            function checkSloganSwitch() {
                const now = Date.now();
                if (now - lastSloganSwitchTime >= SLOGAN_DURATION) {
                    switchSlogan();
                    lastSloganSwitchTime = now;
                }
                requestAnimationFrame(checkSloganSwitch);
            }
            requestAnimationFrame(checkSloganSwitch);
        }

        function switchSlogan() {
            const prevIndex = currentSloganIndex;
            currentSloganIndex = (currentSloganIndex + 1) % sloganItems.length;
            sloganItems[prevIndex].classList.remove('active');
            sloganItems[prevIndex].classList.add('leaving');
            setTimeout(() => {
                sloganItems[prevIndex].classList.remove('leaving');
                sloganItems[currentSloganIndex].classList.add('active');
            }, 100);
        }

        // ==========================================
        // 时间更新系统
        // ==========================================
        function updateTimeLoop() {
            const now = Date.now();
            const targetTime = getTimezoneTime(new Date(now), currentTimezone);
            const currentSecond = targetTime.getSeconds();

            if (currentSecond !== lastSecond) {
                lastSecond = currentSecond;
                let hours = targetTime.getHours();
                const minutes = targetTime.getMinutes().toString().padStart(2, '0');
                const seconds = targetTime.getSeconds().toString().padStart(2, '0');
                let ampm = '';
                if (!is24Hour) {
                    ampm = hours >= 12 ? 'PM' : 'AM';
                    hours = hours % 12;
                    hours = hours ? hours : 12;
                }
                document.getElementById('hours').textContent = hours.toString().padStart(2, '0');
                document.getElementById('minutes').textContent = minutes;
                document.getElementById('seconds').textContent = seconds;
                document.getElementById('ampm').textContent = ampm;

                const year = targetTime.getFullYear();
                const month = (targetTime.getMonth() + 1).toString().padStart(2, '0');
                const day = targetTime.getDate().toString().padStart(2, '0');
                const weekdays = ['周日', '周一', '周二', '周三', '周四', '周五', '周六'];
                const weekday = weekdays[targetTime.getDay()];
                document.getElementById('dateDisplay').textContent = `${year}年${month}月${day}日 ${weekday}`;

                updateSyncCountdown();
                updateExamInfo();

                if (now >= nextSyncTime) {
                    performSync();
                }
            }
            timeAnimationId = requestAnimationFrame(updateTimeLoop);
        }

        function getTimezoneTime(date, timezone) {
            const info = timezoneDB[timezone];
            if (!info) return date;
            const utc = date.getTime() + (date.getTimezoneOffset() * 60000);
            let offset = info.offset;
            if (info.dst && isDST(date, timezone)) offset += 1;
            return new Date(utc + (offset * 3600000));
        }

        function isDST(date, timezone) {
            const info = timezoneDB[timezone];
            if (!info || !info.dst) return false;
            const month = date.getMonth();
            return month > 2 && month < 10;
        }

        function updateSyncCountdown() {
            const now = Date.now();
            const diff = nextSyncTime - now;
            if (diff > 0) {
                const minutes = Math.floor(diff / 60000);
                const seconds = Math.floor((diff % 60000) / 1000);
                document.getElementById('nextSync').textContent = `下次校准: ${minutes.toString().padStart(2,'0')}:${seconds.toString().padStart(2,'0')}`;
            } else {
                document.getElementById('nextSync').textContent = '即将校准...';
            }
        }

        function performSync() {
            const dot = document.getElementById('syncDotInline');
            const text = document.getElementById('syncTextInline');
            const statusRow = document.getElementById('syncStatusInline');
            statusRow.classList.add('syncing');
            text.textContent = '正在同步...';
            setTimeout(() => {
                lastSyncTime = Date.now();
                nextSyncTime = lastSyncTime + 30 * 60 * 1000;
                statusRow.classList.remove('syncing');
                const now = new Date();
                const timeStr = now.getHours().toString().padStart(2,'0') + ':' + now.getMinutes().toString().padStart(2,'0');
                text.textContent = '已同步 ' + timeStr;
                setTimeout(() => { text.textContent = '系统时间正常'; }, 5000);
            }, 1000);
        }

            document.getElementById('examOverlay').classList.remove('active');

        function saveExamSettings() {
            const subject = document.getElementById('examSubjectInput').value.trim();
            const date = document.getElementById('examDate').value;
            const startTime = document.getElementById('examStartTime').value;
            const endTime = document.getElementById('examEndTime').value;

            if (!subject) { alert('请输入考试科目'); return; }
            if (!date || !startTime || !endTime) { alert('请填写完整的考试日期和时间'); return; }

            const [sH, sM] = startTime.split(':').map(Number);
            const [eH, eM] = endTime.split(':').map(Number);
            if (eH * 60 + eM <= sH * 60 + sM) { alert('结束时间必须晚于开始时间'); return; }

            examSettings = { subject, date, startTime, endTime };
            localStorage.setItem('examClock_examSettings', JSON.stringify(examSettings));
            hideExamModal();
            updateExamInfo();
        }

        function clearExamSettings() {
            examSettings = null;
            localStorage.removeItem('examClock_examSettings');
            document.getElementById('examSubjectInput').value = '';
            document.getElementById('examDate').value = '';
            document.getElementById('examStartTime').value = '';
            document.getElementById('examEndTime').value = '';
            hideExamModal();
            updateExamInfo();
        }

        function loadExamSettings() {
            const saved = localStorage.getItem('examClock_examSettings');
            if (saved) {
                examSettings = JSON.parse(saved);
                document.getElementById('examSubjectInput').value = examSettings.subject || '';
                document.getElementById('examDate').value = examSettings.date || '';
                document.getElementById('examStartTime').value = examSettings.startTime || '';
                document.getElementById('examEndTime').value = examSettings.endTime || '';
            } else {
                const now = getTimezoneTime(new Date(), currentTimezone);
                const y = now.getFullYear();
                const m = String(now.getMonth() + 1).padStart(2, '0');
                const d = String(now.getDate()).padStart(2, '0');
                document.getElementById('examDate').value = `${y}-${m}-${d}`;
            }
        }

        function updateExamInfo() {
            const examInfo = document.getElementById('examInfo');
            const subjectDisplay = document.getElementById('examSubjectDisplay');
            const statusDisplay = document.getElementById('examStatus');
            const countdownDisplay = document.getElementById('examCountdown');
            const progressBar = document.getElementById('examProgressBar');

            if (!examSettings) {
                examInfo.style.display = 'none';
                return;
            }

            const now = getTimezoneTime(new Date(), currentTimezone);
            const nowY = now.getFullYear();
            const nowM = now.getMonth();
            const nowD = now.getDate();
            const nowH = now.getHours();
            const nowMin = now.getMinutes();
            const nowSec = now.getSeconds();

            const [sYear, sMonth, sDay] = examSettings.date.split('-').map(Number);
            const [sHour, sMin] = examSettings.startTime.split(':').map(Number);
            const [eHour, eMin] = examSettings.endTime.split(':').map(Number);

            const today = new Date(nowY, nowM, nowD).getTime();
            const examDate = new Date(sYear, sMonth - 1, sDay).getTime();
            const nowMinutes = nowH * 60 + nowMin;
            const startMinutes = sHour * 60 + sMin;
            const endMinutes = eHour * 60 + eMin;
            const totalDuration = endMinutes - startMinutes;

            subjectDisplay.textContent = examSettings.subject;
            examInfo.style.display = 'block';

            if (today < examDate) {
                const diffDays = Math.round((examDate - today) / 86400000);
                statusDisplay.textContent = '即将开始';
                statusDisplay.className = 'exam-status upcoming';
                countdownDisplay.textContent = `还有 ${diffDays} 天`;
                progressBar.style.width = '0%';
            } else if (today > examDate) {
                statusDisplay.textContent = '已结束';
                statusDisplay.className = 'exam-status ended';
                countdownDisplay.textContent = '考试日期已过';
                progressBar.style.width = '100%';
            } else {
                if (nowMinutes < startMinutes) {
                    const diffMin = startMinutes - nowMinutes;
                    const diffH = Math.floor(diffMin / 60);
                    const diffM = diffMin % 60;
                    statusDisplay.textContent = '未开始';
                    statusDisplay.className = 'exam-status upcoming';
                    countdownDisplay.textContent = `距开始 ${diffH}小时${diffM}分`;
                    progressBar.style.width = '0%';
                } else if (nowMinutes >= startMinutes && nowMinutes < endMinutes) {
                    const passedMin = nowMinutes - startMinutes;
                    const remainingMin = endMinutes - nowMinutes;
                    const remainingSec = 60 - nowSec;
                    const progress = (passedMin / totalDuration) * 100;
                    statusDisplay.textContent = '进行中';
                    statusDisplay.className = 'exam-status';
                    const rH = Math.floor(remainingMin / 60);
                    const rM = remainingMin % 60;
                    if (remainingMin > 0) {
                        countdownDisplay.textContent = `剩余 ${rH}小时${rM}分${remainingSec}秒`;
                    } else {
                        countdownDisplay.textContent = `剩余 ${remainingSec}秒`;
                    }
                    progressBar.style.width = `${progress}%`;
                } else {
                    statusDisplay.textContent = '已结束';
                    statusDisplay.className = 'exam-status ended';
                    countdownDisplay.textContent = '考试已结束';
                    progressBar.style.width = '100%';
                }
            }
        }

        // ==========================================
        // 控制功能
        // ==========================================
        function initTimezoneSelect() {
            const list = document.getElementById('timezoneList');
            list.innerHTML = '';
            Object.keys(timezoneDB).forEach(tz => {
                const info = timezoneDB[tz];
                const div = document.createElement('div');
                div.className = 'timezone-option';
                if (tz === currentTimezone) div.classList.add('selected');
                const offsetStr = info.offset >= 0 ? `UTC+${info.offset}` : `UTC${info.offset}`;
                div.innerHTML = `<div style="font-weight: 600; margin-bottom: 4px; font-size: 1rem;">${info.name}</div><div style="font-size: 0.85rem; opacity: 0.6; font-family: var(--font-mono);">${tz} · ${offsetStr}</div>`;
                div.onclick = function() { selectTimezone(tz); };
                list.appendChild(div);
            });
        }

        function selectTimezone(tz) {
            currentTimezone = tz;
            const info = timezoneDB[tz];
            const offsetStr = info.offset >= 0 ? `UTC+${info.offset}` : `UTC${info.offset}`;
            document.getElementById('timezoneDisplay').textContent = `${info.name} (${offsetStr})`;
            document.querySelectorAll('.timezone-option').forEach(el => el.classList.remove('selected'));
            event.currentTarget.classList.add('selected');
            hideTimezoneSelect();
            lastSecond = -1;
            updateExamInfo();
        }

        function showTimezoneSelect() {
            initTimezoneSelect();
            document.getElementById('overlay').classList.add('active');
            document.getElementById('timezoneSelect').classList.add('active');
        }

        function hideTimezoneSelect() {
            document.getElementById('overlay').classList.remove('active');
            document.getElementById('timezoneSelect').classList.remove('active');
        }

        function toggleFormat() {
            is24Hour = !is24Hour;
            const btn = document.getElementById('formatBtn');
            btn.innerHTML = `<span>${is24Hour ? '24H' : '12H'}</span>`;
            btn.classList.toggle('active', is24Hour);
            lastSecond = -1;
        }

        function toggleMode() {
            isDayMode = !isDayMode;
            document.body.classList.toggle('day-mode', isDayMode);
            const btn = document.getElementById('modeBtn');
            btn.innerHTML = `<span>${isDayMode ? '☾ 黑夜' : '☀ 白天'}</span>`;
        }

        function toggleFullscreen() {
            if (!document.fullscreenElement) {
                document.documentElement.requestFullscreen().catch(err => {
                    console.log('全屏请求失败:', err);
                }).then(() => {
                    document.getElementById('fullscreenHint').classList.add('show');
                    setTimeout(() => {
                        document.getElementById('fullscreenHint').classList.remove('show');
                    }, 3000);
                });
            } else {
                document.exitFullscreen();
            }
        }

        let controlsTimeout;
        function showControls() {
            const controls = document.getElementById('controls');
            controls.classList.remove('hidden');
            clearTimeout(controlsTimeout);
            controlsTimeout = setTimeout(() => {
                if (!document.getElementById('timezoneSelect').classList.contains('active') &&
                    !document.getElementById('examModal').classList.contains('active') &&
                    !document.getElementById('privacyModal').classList.contains('active')) {
                    controls.classList.add('hidden');
                }
            }, 5000);
        }

        // 事件监听
        document.addEventListener('keydown', (e) => {
            if (e.key === 'f' || e.key === 'F') toggleFullscreen();
            if (e.key === 'm' || e.key === 'M') toggleMode();
            if (e.key === 't' || e.key === 'T') toggleFormat();
            if (e.key === 'z' || e.key === 'Z') showTimezoneSelect();
            if (e.key === 'e' || e.key === 'E') showExamModal();
            if (e.key === 'p' || e.key === 'P') showPrivacyModal();
            if (e.key === 'Escape') {
                hideTimezoneSelect();
                hideExamModal();
                hidePrivacyModal();
            }
            showControls();
        });

        document.addEventListener('mousemove', showControls);
        document.addEventListener('click', showControls);

        document.addEventListener('fullscreenchange', () => {
            if (!document.fullscreenElement) {
                document.getElementById('fullscreenHint').classList.remove('show');
            }
        });

        window.addEventListener('resize', () => { lastSecond = -1; });

    </script>
