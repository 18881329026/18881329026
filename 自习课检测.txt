<script>
'use strict';

/* ================= 常量与状态 ================= */
var DB_MIN=30, DB_MAX=100;
var LEVELS=[
  {max:45, name:'静 谧', color:'var(--green)',  desc:'理想的自习状态，保持住'},
  {max:60, name:'安 静', color:'#a7d96b',       desc:'合乎自习要求，一切正常'},
  {max:75, name:'偏 吵', color:'var(--orange)', desc:'有交流声，请压低音量'},
  {max:999,name:'嘈 杂', color:'var(--red)',    desc:'已影响他人，请立即安静'}
];
var actx=null, analyser=null, micStream=null, rafId=null;
var running=false, t0=0;
var threshold=60, calibration=94, chimeOn=false;
var needleDb=40, sessionPeak=0;
var quietStart=null, overSince=null, lastOverMs=0, overAccum=0, lastChime=0;
var okMs=0, totalMs=0;
var lastHistPush=0, lastUiUpdate=0, lastLv=-1;
var HIST_N=300, histData=[];
var buf=new Float32Array(2048);
var waveC=null, histC=null;

function $(id){ return document.getElementById(id); }
function toast(msg){
  var t=$('toast'); t.textContent=msg; t.className='show';
  clearTimeout(t._tm); t._tm=setTimeout(function(){ t.className=''; },2400);
}
function fmtTime(ms){
  var s=Math.floor(ms/1000), m=Math.floor(s/60), h=Math.floor(m/60);
  var mm=(m%60<10?'0':'')+(m%60), ss=(s%60<10?'0':'')+(s%60);
  return h>0 ? h+':'+mm+':'+ss : mm+':'+ss;
}
function dbPct(db){
  var t=(db-DB_MIN)/(DB_MAX-DB_MIN);
  return Math.min(1,Math.max(0,t))*100;
}
function levelOf(db){
  for(var i=0;i<4;i++){ if(db<LEVELS[i].max) return i; }
  return 3;
}

/* ================= 启动 / 停止 ================= */
function start(){
  if(!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia){
    toast('当前浏览器不支持麦克风，请使用新版 Chrome / Edge'); return;
  }
  navigator.mediaDevices.getUserMedia({
    audio:{ echoCancellation:false, noiseSuppression:false, autoGainControl:false }
  }).then(function(stream){
    micStream=stream;
    var AC=window.AudioContext||window.webkitAudioContext;
    actx=new AC(); analyser=actx.createAnalyser(); analyser.fftSize=2048;
    actx.createMediaStreamSource(stream).connect(analyser);
    if(actx.state==='suspended'){ actx.resume(); }

    running=true; t0=performance.now(); quietStart=t0;
    sessionPeak=0; okMs=0; totalMs=0; histData.length=0;
    overSince=null; overAccum=0; lastChime=0; lastLv=-1; needleDb=40;

    $('micBtn').style.display='none';
    $('stopBtn').style.display='inline-flex';
    setStatus('on','监听中 · 麦克风正常');
    $('stMic').textContent='监听中';
    fitAll(); drawHist();
    toast('麦克风已接入，监测开始');
    if(rafId) cancelAnimationFrame(rafId);
    rafId=requestAnimationFrame(loop);
  }).catch(function(err){
    var n=err&&err.name?err.name:'';
    if(n==='NotAllowedError') toast('麦克风权限被拒绝，请在地址栏允许后重试');
    else if(n==='NotFoundError') toast('未检测到可用麦克风设备');
    else toast('启动失败：'+n);
  });
}
function stop(){
  running=false;
  if(rafId){ cancelAnimationFrame(rafId); rafId=null; }
  if(micStream){ micStream.getTracks().forEach(function(t){ t.stop(); }); }
  if(actx){ try{ actx.close(); }catch(e){} }
  micStream=null; actx=null; analyser=null;

  $('micBtn').style.display='inline-flex';
  $('stopBtn').style.display='none';
  setStatus('','麦克风未开启');
  $('dbNum').textContent='--.-'; $('dbNum').style.color='';
  $('lvName').textContent='待 机'; $('lvName').style.color='var(--dim)';
  $('lvDesc').textContent='等待接入麦克风';
  $('barFill').style.width='0%';
  $('stMic').textContent='未开启';
  $('waveInfo').textContent='—';
  lastLv=-1; needleDb=40;
  var c=waveC.getContext('2d'); c.clearRect(0,0,waveC.clientWidth,waveC.clientHeight);
}
function setStatus(mode,txt){
  var p=$('statusPill');
  p.className = mode==='bad' ? 'status bad' : (mode==='on'?'status on':'status');
  $('statusTxt').textContent=txt;
}

/* ================= 提示音 ================= */
function chime(){
  if(!actx) return;
  try{
    var t=actx.currentTime, freqs=[880,660];
    for(var i=0;i<freqs.length;i++){
      var o=actx.createOscillator(), g=actx.createGain();
      o.type='triangle'; o.frequency.value=freqs[i];
      var st=t+i*0.38;
      g.gain.setValueAtTime(0.0001,st);
      g.gain.linearRampToValueAtTime(0.22,st+0.03);
      g.gain.exponentialRampToValueAtTime(0.0001,st+0.55);
      o.connect(g); g.connect(actx.destination);
      o.start(st); o.stop(st+0.6);
    }
  }catch(e){}
}

/* ================= 主循环 ================= */
function loop(now){
  if(!running||!analyser) return;
  analyser.getFloatTimeDomainData(buf);

  var sum=0;
  for(var i=0;i<buf.length;i++) sum+=buf[i]*buf[i];
  var rms=Math.sqrt(sum/buf.length);
  var db=Math.max(0,20*Math.log10(Math.max(rms,1e-8))+calibration);

  needleDb += db>needleDb ? (db-needleDb)*0.38 : (db-needleDb)*0.06;
  totalMs+=16.7;

  var lv=levelOf(needleDb);
  var isOver=db>threshold;

  if(isOver){ quietStart=null; }
  else if(quietStart===null){ quietStart=now; }

  if(isOver){
    if(overSince===null){
      if(now-lastOverMs>1000){
        var c=parseInt($('stCounts').textContent,10)||0;
        $('stCounts').textContent=(c+1)+' 次';
      }
      overSince=now; lastOverMs=now;
    }
    overAccum+=16.7;
    if(chimeOn&&overAccum>3000&&now-lastChime>20000){ chime(); lastChime=now; }
  }else{
    overSince=null; overAccum=0; okMs+=16.7;
  }
  if(db>sessionPeak) sessionPeak=db;

  /* --- 读数与颜色 --- */
  var col=isOver?'var(--red)':LEVELS[lv].color;
  $('dbNum').textContent=needleDb.toFixed(1);
  $('dbNum').style.color=col;
  if(lv!==lastLv||isOver){
    $('lvName').textContent=LEVELS[lv].name;
    $('lvName').style.color=col;
    $('lvDesc').textContent=isOver?('已越过报警线 '+threshold+' dB，请安静'):LEVELS[lv].desc;
    lastLv=lv;
  }
  $('barFill').style.width=dbPct(needleDb)+'%';
  $('barFill').style.background=isOver?'var(--red)':(lv>=2?'var(--orange)':'var(--green)');
  setStatus(isOver?'bad':'on', isOver?'音量超标 · 请保持安静':'监听中 · 麦克风正常');

  drawWave(isOver);

  if(now-lastHistPush>200){
    lastHistPush=now;
    histData.push(db);
    if(histData.length>HIST_N) histData.shift();
    drawHist();
  }
  if(now-lastUiUpdate>250){
    lastUiUpdate=now;
    $('stQuiet').textContent=quietStart?fmtTime(now-quietStart):'00:00';
    $('stPeak').textContent=(sessionPeak>0?sessionPeak.toFixed(1):'--')+' dB';
    $('stPeak').className=sessionPeak>threshold?'warn':'';
    $('stRate').textContent=(totalMs>2000?Math.round(okMs/totalMs*100):'--')+'%';
    $('waveInfo').textContent=actx?('采样 '+actx.sampleRate+' Hz · 校准 +'+calibration+' dB'):'—';
  }
  rafId=requestAnimationFrame(loop);
}

/* ================= Canvas ================= */
function fitCanvas(c){
  var dpr=window.devicePixelRatio||1;
  c.width=c.clientWidth*dpr; c.height=c.clientHeight*dpr;
  c.getContext('2d').setTransform(dpr,0,0,dpr,0,0);
}
function fitAll(){ if(waveC)fitCanvas(waveC); if(histC)fitCanvas(histC); }

function drawWave(isOver){
  var ctx=waveC.getContext('2d'), w=waveC.clientWidth, h=waveC.clientHeight;
  ctx.clearRect(0,0,w,h);
  ctx.strokeStyle=isOver?'var(--red)':'rgba(128,128,128,.25)';
  ctx.strokeStyle=isOver?'rgba(248,113,113,.9)':'rgba(128,128,128,.35)';
  ctx.lineWidth=1.2; ctx.beginPath();
  var n=buf.length, step=Math.max(1,Math.floor(n/w));
  for(var x=0;x<w;x++){
    var v=buf[Math.min(n-1,x*step)];
    var y=h/2-v*h*0.45;
    if(x===0)ctx.moveTo(x,y); else ctx.lineTo(x,y);
  }
  ctx.stroke();
  ctx.strokeStyle='rgba(128,128,128,.15)'; ctx.lineWidth=1;
  ctx.beginPath(); ctx.moveTo(0,h/2); ctx.lineTo(w,h/2); ctx.stroke();
}
function drawHist(){
  var ctx=histC.getContext('2d'), w=histC.clientWidth, h=histC.clientHeight;
  ctx.clearRect(0,0,w,h);
  var bw=w/HIST_N;
  function yOf(db){ return h-Math.min(1,Math.max(0,(db-DB_MIN)/(DB_MAX-DB_MIN)))*h; }
  var ty=Math.round(yOf(threshold))+0.5;
  ctx.strokeStyle='rgba(248,113,113,.6)'; ctx.setLineDash([4,4]);
  ctx.beginPath(); ctx.moveTo(0,ty); ctx.lineTo(w,ty); ctx.stroke(); ctx.setLineDash([]);
  var off=HIST_N-histData.length;
  for(var i=0;i<histData.length;i++){
    var x=(off+i)*bw, y=yOf(histData[i]);
    ctx.fillStyle=histData[i]<threshold?'rgba(74,222,128,.7)':'rgba(248,113,113,.85)';
    ctx.fillRect(x+0.3,y,Math.max(1,bw-0.6),h-y);
  }
}

    closes[i].addEventListener('click',function(){
      $(this.getAttribute('data-close')).className='modal';
    });
  }
  var oms=document.querySelectorAll('.modal');
  for(var j=0;j<oms.length;j++){
    oms[j].addEventListener('click',function(e){ if(e.target===this) this.className='modal'; });
  }

    threshold=+e.target.value;
    $('thVal').textContent=threshold+' dB';
    $('thMark').style.left=dbPct(threshold)+'%';
    lastLv=-1;
  });
  $('calSlider').addEventListener('input',function(e){
    calibration=+e.target.value;
    $('calVal').textContent='+'+calibration+' dB';
  });
  $('bellBtn').addEventListener('click',function(){
    chimeOn=!chimeOn;
    this.className=chimeOn?'tbtn active':'tbtn';
    toast(chimeOn?'提示音已开启：持续超标 3 秒后响铃':'提示音已关闭');

    var day=document.body.className==='day';

  });

    if(document.fullscreenElement){ document.exitFullscreen(); }
    else{
      var p=document.documentElement.requestFullscreen();
      if(p&&p.catch) p.catch(function(){ toast('当前环境不支持全屏'); });

    $('escHint').style.display=document.fullscreenElement?'block':'none';

    }
  });
}
